cmake_minimum_required(VERSION 3.20)
project(abcore CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(abcore main.cpp)
# Bitre azonos lebegőpontos eredmény a Python/C# változattal: nincs FMA-összevonás, nincs fast-math.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(abcore PRIVATE -ffp-contract=off -fno-fast-math -Wall)
endif()
if(WIN32)
  target_link_libraries(abcore PRIVATE psapi)
endif()
# MinGW: statikus futásidő, hogy ne egy idegen (pl. a Git-ből betöltött) libstdc++ DLL fusson
if(MINGW)
  target_link_options(abcore PRIVATE -static -static-libgcc -static-libstdc++)
endif()
